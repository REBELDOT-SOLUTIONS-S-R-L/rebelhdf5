export function humanizeColumnName(column: string): string {
  let label = column;
  for (const prefix of ['dist_', 'threshold_', 'keypoint_', 'eef_']) {
    label = label.replaceAll(prefix, '');
  }
  label = label.replaceAll('_m', '');
  label = label.replaceAll('_', ' ');
  return label;
}
